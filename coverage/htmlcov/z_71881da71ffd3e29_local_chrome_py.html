1"""Local Chrome bridge for attaching chrome-devtools-mcp to a host browser. 

2 

3Probes the Windows host's Chrome debug port from inside the container and 

4writes a minimal MCP config JSON that Claude Code can consume. 

5 

6Chrome's DevTools Protocol rejects HTTP requests whose ``Host`` header is 

7not ``localhost`` or an IP address. Because the MCP server inside the 

8container connects via ``host.docker.internal``, Chrome returns HTTP 500. 

9To work around this, we start a small Node.js TCP proxy inside the 

10container that forwards ``localhost:<port>`` to ``host.docker.internal:<port>``. 

11The MCP server then connects to ``localhost:<port>`` and Chrome sees a 

12``Host: localhost`` header. 

13""" 

14

86def find_chrome() -> str | None: 

87 """Locate the Chrome executable on the host. 

88 

89 Windows: well-known install paths, then the per-user install. macOS: the 

90 app bundles. Linux: Chrome or Chromium on PATH. Returns the path as a 

91 string or ``None`` if Chrome is not found. 

92 """ 

93 system = platform.system() 

94 if system == "Windows": 

95 for candidate in _CHROME_CANDIDATES: 

96 if Path(candidate).exists(): 

97 return candidate 

98 # Fallback: per-user Chrome install 

99 local_app = os.environ.get("LOCALAPPDATA", "") 

100 if local_app: 

101 user_chrome = Path(local_app) / "Google" / "Chrome" / "Application" / "chrome.exe" 

102 if user_chrome.exists(): 

103 return str(user_chrome) 

104 return None 

105 if system == "Darwin": 

106 for candidate in _CHROME_MAC_CANDIDATES: 

107 path = Path(candidate).expanduser() 

108 if path.exists(): 

109 return str(path) 

110 return None 

111 for name in _CHROME_LINUX_CANDIDATES: 

112 found = shutil.which(name) 

113 if found: 

114 return found 

115 return None

118def chrome_bind_address(container_name: str) -> str: 

119 """Address Chrome's debug port must listen on so the container can reach it. 

120 

121 Docker Desktop (Windows, macOS) forwards ``host.docker.internal`` to the 

122 host's loopback, so ``127.0.0.1`` works there. On a Linux host (Docker or 

123 podman) ``host.docker.internal`` resolves to the bridge gateway address, 

124 and a loopback-only listener is unreachable from containers, so Chrome 

125 binds to the address the container actually resolves for the host. That 

126 address is reachable from the host and its containers only. 

127 """ 

128 if platform.system() != "Linux": 

129 return LOOPBACK 

130 result = subprocess.run( 

131 ["docker", "exec", container_name, "getent", "hosts", CHROME_DEBUG_HOST], 

132 capture_output=True, 

133 text=True, 

134 ) 

135 if result.returncode == 0 and result.stdout.strip(): 

136 candidate = result.stdout.split()[0] 

137 try: 

138 ipaddress.ip_address(candidate) 

139 except ValueError: 

140 logger.debug("Unexpected %s resolution %r", CHROME_DEBUG_HOST, candidate) 

141 else: 

142 return candidate 

143 logger.warning( 

144 "%s does not resolve inside %s; Chrome will bind to loopback, which Linux " 

145 "containers cannot reach. Docker needs `host-gateway` support (20.10+), podman 4.1+.", 

146 CHROME_DEBUG_HOST, 

147 container_name, 

148 ) 

149 return LOOPBACK

152def _project_slug(project_name: str, project_dir: str | Path | None = None) -> str: 

153 """Return a stable slug for project-scoped Chrome state.""" 

154 if project_dir is not None: 

155 try: 

156 return unique_project_name(Path(project_dir), project_name) 

157 except (TypeError, ValueError): 

158 logger.debug("Falling back to project-name-only slug for %s", project_name) 

159 slug = "".join(ch if ch.isalnum() or ch == "-" else "-" for ch in project_name.lower()) 

160 slug = "-".join(part for part in slug.split("-") if part) 

161 return slug or "project"

164def _chrome_slug( 

165 project_name: str, 

166 project_dir: str | Path | None = None, 

167 profile: str | None = None, 

168) -> str: 

169 """Slug that keys a Chrome instance: per account when *profile* is set, else per project. 

170 

171 Logged-in browser state follows a person, not a repo, so a selected Claude 

172 account gets one Chrome (profile dir + debug port) shared by every project 

173 that uses that account. Two accounts never share a window. 

174 """ 

175 if profile: 

176 slug = "".join(ch if ch.isalnum() or ch == "-" else "-" for ch in profile.lower()) 

177 slug = "-".join(part for part in slug.split("-") if part) 

178 return _ACCOUNT_PROFILE_PREFIX + (slug or "default") 

179 return _project_slug(project_name, project_dir)

196 project_name: str,

200 """Return a stable Chrome remote debugging port for the project or account."""

265def launch_chrome( 

266 port: int, 

267 *, 

268 project_name: str, 

269 project_dir: str | Path | None = None, 

270 profile: str | None = None, 

271 bind_address: str = LOOPBACK, 

272 proxy: str | None = None, 

273) -> bool: 

274 """Launch Chrome on the host with the debug port enabled. 

275 

276 Returns ``True`` if Chrome was launched, ``False`` if Chrome could 

277 not be found. The process is started detached so it outlives the CLI. 

278 With *proxy*, Chrome sends its traffic through that proxy server (a VPN 

279 container's HTTP proxy, for example). 

280 """ 

281 chrome_path = find_chrome() 

282 if chrome_path is None: 

283 return False 

284 

285 profile_dir = _chrome_profile_dir(project_name, project_dir, profile) 

286 args = [ 

287 chrome_path, 

288 f"--remote-debugging-port={port}", 

289 f"--remote-debugging-address={bind_address}", 

290 "--remote-allow-origins=*", 

291 f"--user-data-dir={profile_dir}", 

292 # A fresh profile dir otherwise opens first-run / default-browser dialogs. 

293 "--no-first-run", 

294 "--no-default-browser-check", 

295 *_proxy_args(proxy), 

296 ] 

297 logger.info("Launching Chrome: %s", " ".join(args)) 

298 _spawn_detached(args) 

299 return True

302def chrome_profile_in_use(profile_dir: str | Path) -> bool: 

303 """True when a Chrome process holds *profile_dir* open. 

304 

305 Chrome on Linux and macOS leaves a ``SingletonLock`` link in the profile 

306 while it runs and removes it on exit. Windows keeps its ``lockfile`` on 

307 disk, so there the answer is always ``False`` and the caller relies on 

308 the debug port probe instead. 

309 """ 

310 if platform.system() == "Windows": 

311 return False 

312 return Path(profile_dir, "SingletonLock").is_symlink()

315def launch_plain_chrome( 

316 *, 

317 project_name: str, 

318 project_dir: str | Path | None = None, 

319 profile: str | None = None, 

320 proxy: str | None = None, 

321 url: str | None = None, 

322) -> str: 

323 """Open the project's (or account's) ai-shell Chrome WITHOUT the debug port. 

324 

325 Google refuses to sign in inside a browser that exposes a DevTools port 

326 ("This browser or app may not be secure"), so a Google (or Google SSO) 

327 sign-in has to happen in a plain window on the same profile directory. 

328 The cookies it leaves behind are what the debug-port Chrome reuses on 

329 the next ``--local-chrome`` launch. Returns the profile directory. 

330 

331 Raises :class:`LocalChromeUnavailable` when Chrome is not installed, or 

332 when a Chrome already holds that profile: a second launch would only 

333 open a window inside the running (debug-port) process. 

334 """ 

335 chrome_path = find_chrome() 

336 if chrome_path is None: 

337 raise LocalChromeUnavailable( 

338 "Chrome could not be found. Install Google Chrome or Chromium, or put it on PATH." 

339 ) 

340 profile_dir = _chrome_profile_dir(project_name, project_dir, profile) 

341 port = _project_debug_port(project_name, project_dir, profile) 

342 if chrome_profile_in_use(profile_dir) or probe_host_chrome_port(port): 

343 raise LocalChromeUnavailable( 

344 f"A Chrome window already uses the profile {profile_dir}.\n" 

345 "Close every window of that Chrome (the one ai-shell opened with the " 

346 "DevTools port) and run this again: a second launch would open inside " 

347 "the running process, where Google still refuses the sign-in." 

348 ) 

349 args = [ 

350 chrome_path, 

351 f"--user-data-dir={profile_dir}", 

352 "--no-first-run", 

353 "--no-default-browser-check", 

354 *_proxy_args(proxy), 

355 ] 

356 if url: 

357 args.append(url) 

358 logger.info("Launching plain Chrome: %s", " ".join(args)) 

359 _spawn_detached(args) 

360 return profile_dir

363def probe_chrome_port(container_name: str, port: int) -> bool: 

364 """Check whether a Chrome debug port is reachable from the container. 

365 

366 Returns ``True`` if reachable, ``False`` otherwise. 

367 """ 

368 probe_url = f"http://{CHROME_DEBUG_HOST}:{port}/json/version" 

369 args = [ 

370 "docker", 

371 "exec", 

372 container_name, 

373 "curl", 

374 "-sS", 

375 "--max-time", 

376 "3", 

377 "-H", 

378 f"Host: localhost:{port}", 

379 probe_url, 

380 ] 

381 logger.debug("Probing Chrome debug port: %s", " ".join(args)) 

382 result = subprocess.run(args, capture_output=True, text=True) 

383 if result.returncode != 0 or not result.stdout.strip(): 

384 return False 

385 logger.info("Chrome debug port reachable: %s", result.stdout.strip()[:120]) 

386 return True

389def probe_host_chrome_port(port: int, host: str = LOOPBACK) -> bool: 

390 """Check whether a Chrome debug port is reachable on the host at *host*.""" 

391 connection = HTTPConnection(host, port, timeout=2) 

392 try: 

393 connection.request("GET", "/json/version") 

394 response = connection.getresponse() 

395 return response.status == 200 and bool(response.read().strip()) 

396 except (OSError, HTTPException): 

397 return False 

398 finally: 

399 connection.close()

402def _wait_until_ready( 

403 probe_fn: Callable[..., bool], 

404 *args: object, 

405 timeout_seconds: float, 

406 interval_seconds: float = CHROME_PROBE_INTERVAL_SECONDS, 

407) -> bool: 

408 """Poll until a probe succeeds or the timeout expires.""" 

409 deadline = time.monotonic() + timeout_seconds 

410 while True: 

411 if probe_fn(*args): 

412 return True 

413 remaining = deadline - time.monotonic() 

414 if remaining <= 0: 

415 return False 

416 time.sleep(min(interval_seconds, remaining))

419def ensure_host_chrome( 

420 container_name: str, 

421 *, 

422 project_name: str, 

423 project_dir: str | Path | None = None, 

424 profile: str | None = None, 

425 proxy: str | None = None, 

426) -> int: 

427 """Ensure Chrome is running with a debug port reachable from the container. 

428 

429 Each project gets its own debug profile directory and a stable debug port, 

430 so different repos can keep separate logged-in Chrome instances alive. 

431 When *profile* is given (a Claude account), the profile and port are keyed 

432 by it instead, so one account keeps one browser across projects and two 

433 accounts never share one. *proxy* is handed to Chrome as 

434 ``--proxy-server`` when ai-shell has to launch it; a Chrome that is 

435 already running keeps whatever proxy it was started with. 

436 

437 Returns the port number Chrome is listening on. 

438 """ 

439 port = _project_debug_port(project_name, project_dir, profile) 

440 profile_dir = _chrome_profile_dir(project_name, project_dir, profile) 

441 

442 if probe_chrome_port(container_name, port): 

443 return port 

444 

445 logger.info("Chrome for project %s not found on port %d, launching it", project_name, port) 

446 

447 bind_address = chrome_bind_address(container_name) 

448 if not launch_chrome( 

449 port, 

450 project_name=project_name, 

451 project_dir=project_dir, 

452 profile=profile, 

453 bind_address=bind_address, 

454 proxy=proxy, 

455 ): 

456 raise LocalChromeUnavailable( 

457 _build_setup_instructions(project_name, profile_dir, port, bind_address, proxy) 

458 ) 

459 

460 if not _wait_until_ready( 

461 probe_host_chrome_port, 

462 port, 

463 bind_address, 

464 timeout_seconds=CHROME_HOST_PROBE_TIMEOUT_SECONDS, 

465 ): 

466 raise LocalChromeUnavailable( 

467 f"Chrome was launched for project '{project_name}' on port {port}, but the " 

468 f"debug port did not open on {bind_address} within " 

469 f"{int(CHROME_HOST_PROBE_TIMEOUT_SECONDS)} seconds.\n\n" 

470 "If another ai-shell Chrome window for this project is already open, " 

471 "close it and retry." 

472 ) 

473 

474 if _wait_until_ready( 

475 probe_chrome_port, 

476 container_name, 

477 port, 

478 timeout_seconds=CHROME_CONTAINER_PROBE_TIMEOUT_SECONDS, 

479 ): 

480 logger.info("Chrome ready on port %d for project %s", port, project_name) 

481 return port 

482 

483 raise LocalChromeUnavailable( 

484 f"Chrome is listening on {bind_address}:{port} for project '{project_name}', but the " 

485 "debug port did not become reachable from the dev container within " 

486 f"{int(CHROME_CONTAINER_PROBE_TIMEOUT_SECONDS)} seconds.\n\n" 

487 "Check that the container can reach the host via host.docker.internal " 

488 "(Docker Desktop on Windows/macOS; `host-gateway` on Linux Docker 20.10+ or podman 4.1+)." 

489 )

492def start_chrome_proxy(container_name: str, port: int) -> None: 

493 """Start a TCP proxy inside the container: localhost:<port> -> host.docker.internal:<port>. 

494 

495 Chrome rejects DevTools Protocol requests with a non-localhost Host 

496 header. This proxy lets the MCP server connect to ``localhost:<port>`` 

497 so Chrome sees ``Host: localhost``. 

498 

499 The proxy runs as a detached background process via ``docker exec -d``. 

500 It's idempotent -- if the port is already in use (previous proxy still 

501 running), the new one fails silently and the existing one keeps working. 

502 """ 

503 script = _NODE_PROXY_TEMPLATE.format(port=port) 

504 args = [ 

505 "docker", 

506 "exec", 

507 "-d", 

508 container_name, 

509 "node", 

510 "-e", 

511 script, 

512 ] 

513 logger.debug("Starting Chrome proxy: %s", " ".join(args)) 

514 result = subprocess.run(args, capture_output=True, text=True) 

515 if result.returncode != 0: 

516 logger.warning( 

517 "Chrome proxy start returned %d: %s", result.returncode, result.stderr.strip() 

518 )

541def write_mcp_config(port: int, config_dir: Path | None = None) -> Path: 

542 """Write the chrome-devtools-mcp server config JSON. 

543 

544 Returns the path to the written file. The file lives under

557 return path

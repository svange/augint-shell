1"""Local Chrome bridge for attaching chrome-devtools-mcp to a host browser. 

2 

3Probes the Windows host's Chrome debug port from inside the container and 

4writes a minimal MCP config JSON that Claude Code can consume. 

5 

6Chrome's DevTools Protocol rejects HTTP requests whose ``Host`` header is 

7not ``localhost`` or an IP address. Because the MCP server inside the 

8container connects via ``host.docker.internal``, Chrome returns HTTP 500. 

9To work around this, we start a small Node.js TCP proxy inside the 

10container that forwards ``localhost:<port>`` to ``host.docker.internal:<port>``. 

11The MCP server then connects to ``localhost:<port>`` and Chrome sees a 

12``Host: localhost`` header. 

13"""

94def find_chrome() -> str | None: 

95 """Locate the Chrome executable on the host. 

96 

97 Windows: well-known install paths, then the per-user install. macOS: the 

98 app bundles. Linux: Chrome or Chromium on PATH. Returns the path as a 

99 string or ``None`` if Chrome is not found. 

100 """ 

101 system = platform.system() 

102 if system == "Windows": 

103 for candidate in _CHROME_CANDIDATES: 

104 if Path(candidate).exists(): 

105 return candidate 

106 # Fallback: per-user Chrome install 

107 local_app = os.environ.get("LOCALAPPDATA", "") 

108 if local_app: 

109 user_chrome = Path(local_app) / "Google" / "Chrome" / "Application" / "chrome.exe" 

110 if user_chrome.exists(): 

111 return str(user_chrome) 

112 return None 

113 if system == "Darwin": 

114 for candidate in _CHROME_MAC_CANDIDATES: 

115 path = Path(candidate).expanduser() 

116 if path.exists(): 

117 return str(path) 

118 return None 

119 for name in _CHROME_LINUX_CANDIDATES: 

120 found = shutil.which(name) 

121 if found: 

122 return found 

123 return None

126def chrome_bind_address(container_name: str) -> str: 

127 """Address the container reaches the host at, where the DevTools port must answer. 

128 

129 Docker Desktop (Windows, macOS) forwards ``host.docker.internal`` to the 

130 host's loopback, so ``127.0.0.1`` works there. On a Linux host (Docker or 

131 podman) ``host.docker.internal`` resolves to a real host address, and a 

132 loopback-only listener is unreachable from containers. A windowed Chrome 

133 binds to loopback only (it ignores ``--remote-debugging-address``), so on 

134 that address ai-shell runs :mod:`ai_shell.cdp_relay`, which forwards to 

135 Chrome's loopback port and serves the host's own addresses only. 

136 """ 

137 if platform.system() != "Linux": 

138 return LOOPBACK 

139 result = subprocess.run( 

140 ["docker", "exec", container_name, "getent", "hosts", CHROME_DEBUG_HOST], 

141 capture_output=True, 

142 text=True, 

143 ) 

144 if result.returncode == 0 and result.stdout.strip(): 

145 candidate = result.stdout.split()[0] 

146 try: 

147 ipaddress.ip_address(candidate) 

148 except ValueError: 

149 logger.debug("Unexpected %s resolution %r", CHROME_DEBUG_HOST, candidate) 

150 else: 

151 return candidate 

152 logger.warning( 

153 "%s does not resolve inside %s; only Chrome's loopback port exists, which Linux " 

154 "containers cannot reach. Docker needs `host-gateway` support (20.10+), podman 4.1+.", 

155 CHROME_DEBUG_HOST, 

156 container_name, 

157 ) 

158 return LOOPBACK

308def _wait_until_profile_free(profile_dir: str | Path, port: int, timeout: float) -> bool: 

309 """Wait until no Chrome holds *profile_dir* (lock gone, debug port closed).""" 

310 deadline = time.monotonic() + timeout 

311 while time.monotonic() < deadline: 

312 if not chrome_profile_in_use(profile_dir) and not probe_host_chrome_port(port): 

313 return True 

314 time.sleep(0.5) 

315 return not chrome_profile_in_use(profile_dir) and not probe_host_chrome_port(port)

318def close_chrome_profile(profile_dir: str | Path, port: int) -> bool: 

319 """Ask the Chrome that holds *profile_dir* to quit, and wait for it. 

320 

321 The window is ai-shell's own (its profile directory is under the 

322 ``ai-shell`` folder), so closing it is in scope. The request is a 

323 graceful one (SIGTERM on POSIX, ``taskkill`` without ``/F`` on Windows), so 

324 Chrome flushes its cookies: the whole point is to keep the session for 

325 the next launch. Returns ``True`` when the profile is free afterwards. 

326 """ 

327 profile_dir = str(profile_dir) 

328 if platform.system() == "Windows": 

329 script = ( 

330 "Get-CimInstance Win32_Process -Filter \"Name='chrome.exe'\" | " 

331 f"Where-Object {{ $_.CommandLine -like '*--user-data-dir={profile_dir}*' }} | " 

332 "ForEach-Object { taskkill /PID $_.ProcessId | Out-Null }" 

333 ) 

334 args = ["powershell", "-NoProfile", "-NonInteractive", "-Command", script] 

335 else: 

336 args = ["pkill", "-TERM", "-f", "--", f"--user-data-dir={re.escape(profile_dir)}"] 

337 logger.info("Closing Chrome on profile %s", profile_dir) 

338 subprocess.run(args, capture_output=True, text=True) # noqa: S603 

339 return _wait_until_profile_free(profile_dir, port, CHROME_CLOSE_TIMEOUT_SECONDS)

342def launch_chrome( 

343 port: int, 

344 *, 

345 project_name: str, 

346 project_dir: str | Path | None = None, 

347 profile: str | None = None, 

348 bind_address: str = LOOPBACK, 

349 proxy: str | None = None, 

350) -> bool: 

351 """Launch Chrome on the host with the debug port enabled. 

352 

353 Returns ``True`` if Chrome was launched, ``False`` if Chrome could 

354 not be found. The process is started detached so it outlives the CLI. 

355 With *proxy*, Chrome sends its traffic through that proxy server (a VPN 

356 container's HTTP proxy, for example). 

357 """ 

358 chrome_path = find_chrome() 

359 if chrome_path is None: 

360 return False 

361 

362 profile_dir = _chrome_profile_dir(project_name, project_dir, profile) 

363 # No --remote-debugging-address: a windowed Chrome ignores it and binds to 

364 # loopback regardless; *bind_address* is served by the relay instead. 

365 del bind_address 

366 args = [ 

367 chrome_path, 

368 f"--remote-debugging-port={port}", 

369 "--remote-allow-origins=*", 

370 f"--user-data-dir={profile_dir}", 

371 # A fresh profile dir otherwise opens first-run / default-browser dialogs. 

372 "--no-first-run", 

373 "--no-default-browser-check", 

374 *_proxy_args(proxy), 

375 ] 

376 logger.info("Launching Chrome: %s", " ".join(args)) 

377 _spawn_detached(args) 

378 return True

381def chrome_profile_in_use(profile_dir: str | Path) -> bool: 

382 """True when a Chrome process holds *profile_dir* open. 

383 

384 Chrome on Linux and macOS leaves a ``SingletonLock`` link in the profile 

385 while it runs and removes it on exit. Windows keeps its ``lockfile`` on 

386 disk, so there the answer is always ``False`` and the caller relies on 

387 the debug port probe instead. 

388 """ 

389 if platform.system() == "Windows": 

390 return False 

391 return Path(profile_dir, "SingletonLock").is_symlink()

395class PlainChrome: 

396 """A Chrome window ai-shell opened without the DevTools port."""

402 def close(self) -> bool: 

403 """Quit the window gracefully and wait until the profile is free.""" 

404 if self.process.poll() is None: 

405 if platform.system() == "Windows":

419def launch_plain_chrome( 

420 *, 

421 project_name: str, 

422 project_dir: str | Path | None = None, 

423 profile: str | None = None, 

424 proxy: str | None = None, 

425 url: str | None = None, 

426 close_running: bool = False, 

427) -> PlainChrome: 

428 """Open the project's (or account's) ai-shell Chrome WITHOUT the debug port. 

429 

430 Google refuses to sign in inside a browser that exposes a DevTools port 

431 ("This browser or app may not be secure"), so a Google (or Google SSO) 

432 sign-in has to happen in a plain window on the same profile directory. 

433 The cookies it leaves behind are what the debug-port Chrome reuses on 

434 the next ``--local-chrome`` launch. 

435 

436 Raises :class:`LocalChromeUnavailable` when Chrome is not installed, or 

437 when a Chrome already holds that profile (a second launch would only 

438 open a window inside the running, debug-port process). With 

439 *close_running*, that Chrome is asked to quit first: it is ai-shell's own 

440 window. 

441 """ 

442 chrome_path = find_chrome() 

443 if chrome_path is None: 

444 raise LocalChromeUnavailable( 

445 "Chrome could not be found. Install Google Chrome or Chromium, or put it on PATH." 

446 ) 

447 profile_dir = _chrome_profile_dir(project_name, project_dir, profile) 

448 port = _project_debug_port(project_name, project_dir, profile) 

449 if chrome_profile_in_use(profile_dir) or probe_host_chrome_port(port): 

450 if not (close_running and close_chrome_profile(profile_dir, port)): 

451 raise LocalChromeUnavailable( 

452 f"A Chrome window already uses the profile {profile_dir}.\n" 

453 "Close every window of that Chrome (the one ai-shell opened with the " 

454 "DevTools port) and run this again: a second launch would open inside " 

455 "the running process, where Google still refuses the sign-in." 

456 ) 

457 reason = no_display_reason() 

458 if reason: 

459 raise LocalChromeUnavailable(reason) 

460 args = [ 

461 chrome_path, 

462 f"--user-data-dir={profile_dir}", 

463 "--no-first-run", 

464 "--no-default-browser-check", 

465 *_proxy_args(proxy), 

466 ] 

467 if url: 

468 args.append(url) 

469 logger.info("Launching plain Chrome: %s", " ".join(args)) 

470 # Chrome's stderr goes to a temp file: when it exits at once, its own 

471 # words ("Missing X server or $DISPLAY", a profile error) are the reason. 

472 stderr = tempfile.TemporaryFile() 

473 process = _spawn_detached(args, stderr=stderr) 

474 code = _exited_early(process, CHROME_START_SECONDS) 

475 if code is not None: 

476 stderr.seek(0) 

477 output = stderr.read().decode("utf-8", "replace").strip() 

478 stderr.close() 

479 tail = "\n".join(output.splitlines()[-8:]) or "(no output)" 

480 raise LocalChromeUnavailable( 

481 f"Chrome exited at once (code {code}) instead of opening the window. Its output:\n{tail}" 

482 ) 

483 stderr.close() 

484 return PlainChrome(profile_dir=profile_dir, port=port, process=process)

487def probe_chrome_port(container_name: str, port: int) -> bool: 

488 """Check whether a Chrome debug port is reachable from the container. 

489 

490 Returns ``True`` if reachable, ``False`` otherwise. 

491 """ 

492 probe_url = f"http://{CHROME_DEBUG_HOST}:{port}/json/version" 

493 args = [ 

494 "docker", 

495 "exec", 

496 container_name, 

497 "curl", 

498 "-sS", 

499 "--max-time", 

500 "3", 

501 "-H", 

502 f"Host: localhost:{port}", 

503 probe_url, 

504 ] 

505 logger.debug("Probing Chrome debug port: %s", " ".join(args)) 

506 result = subprocess.run(args, capture_output=True, text=True) 

507 if result.returncode != 0 or not result.stdout.strip(): 

508 return False 

509 logger.info("Chrome debug port reachable: %s", result.stdout.strip()[:120]) 

510 return True

518def start_cdp_relay(bind_address: str, port: int) -> None: 

519 """Relay *bind_address*:*port* to Chrome's loopback port, detached, unless one answers already. 

520 

521 The relay is :mod:`ai_shell.cdp_relay`, run with the same interpreter 

522 that runs ai-shell; it exits by itself once Chrome's port is gone. 

523 """ 

524 if probe_host_chrome_port(port, bind_address): 

525 return 

526 log_path = relay_log_path(port) 

527 log_path.parent.mkdir(parents=True, exist_ok=True) 

528 args = [sys.executable, "-m", "ai_shell.cdp_relay", bind_address, str(port)] 

529 logger.info("Starting the DevTools relay: %s", " ".join(args)) 

530 with log_path.open("ab") as log: 

531 _spawn_detached(args, stderr=log)

534def probe_host_chrome_port(port: int, host: str = LOOPBACK) -> bool: 

535 """Check whether a Chrome debug port is reachable on the host at *host*.""" 

536 connection = HTTPConnection(host, port, timeout=2) 

537 try: 

538 connection.request("GET", "/json/version") 

539 response = connection.getresponse() 

540 return response.status == 200 and bool(response.read().strip()) 

541 except (OSError, HTTPException): 

542 return False 

543 finally: 

544 connection.close()

547def _wait_until_ready( 

548 probe_fn: Callable[..., bool], 

549 *args: object, 

550 timeout_seconds: float, 

551 interval_seconds: float = CHROME_PROBE_INTERVAL_SECONDS, 

552) -> bool: 

553 """Poll until a probe succeeds or the timeout expires.""" 

554 deadline = time.monotonic() + timeout_seconds 

555 while True: 

556 if probe_fn(*args): 

557 return True 

558 remaining = deadline - time.monotonic() 

559 if remaining <= 0: 

560 return False 

561 time.sleep(min(interval_seconds, remaining))

564def ensure_host_chrome( 

565 container_name: str, 

566 *, 

567 project_name: str, 

568 project_dir: str | Path | None = None, 

569 profile: str | None = None, 

570 proxy: str | None = None, 

571) -> int: 

572 """Ensure Chrome is running with a debug port reachable from the container. 

573 

574 Each project gets its own debug profile directory and a stable debug port, 

575 so different repos can keep separate logged-in Chrome instances alive. 

576 When *profile* is given (a Claude account), the profile and port are keyed 

577 by it instead, so one account keeps one browser across projects and two 

578 accounts never share one. *proxy* is handed to Chrome as 

579 ``--proxy-server`` when ai-shell has to launch it; a Chrome that is 

580 already running keeps whatever proxy it was started with. 

581 

582 Returns the port number Chrome is listening on. 

583 """ 

584 port = _project_debug_port(project_name, project_dir, profile) 

585 profile_dir = _chrome_profile_dir(project_name, project_dir, profile) 

586 

587 if probe_chrome_port(container_name, port): 

588 return port 

589 

590 logger.info("Chrome for project %s not found on port %d, launching it", project_name, port) 

591 

592 bind_address = chrome_bind_address(container_name) 

593 if not launch_chrome( 

594 port, 

595 project_name=project_name, 

596 project_dir=project_dir, 

597 profile=profile, 

598 bind_address=bind_address, 

599 proxy=proxy, 

600 ): 

601 raise LocalChromeUnavailable( 

602 _build_setup_instructions(project_name, profile_dir, port, bind_address, proxy) 

603 ) 

604 

605 if not _wait_until_ready( 

606 probe_host_chrome_port, 

607 port, 

608 LOOPBACK, 

609 timeout_seconds=CHROME_HOST_PROBE_TIMEOUT_SECONDS, 

610 ): 

611 raise LocalChromeUnavailable( 

612 f"Chrome was launched for project '{project_name}' on port {port}, but the " 

613 f"debug port did not open on {LOOPBACK} within " 

614 f"{int(CHROME_HOST_PROBE_TIMEOUT_SECONDS)} seconds.\n\n" 

615 "If another ai-shell Chrome window for this project is already open, " 

616 "close it and retry." 

617 ) 

618 

619 if bind_address != LOOPBACK: 

620 # Linux: the container reaches the host at bind_address, where only 

621 # the relay can answer (Chrome itself listens on loopback alone). 

622 start_cdp_relay(bind_address, port) 

623 if not _wait_until_ready( 

624 probe_host_chrome_port, 

625 port, 

626 bind_address, 

627 timeout_seconds=CHROME_HOST_PROBE_TIMEOUT_SECONDS, 

628 ): 

629 raise LocalChromeUnavailable( 

630 f"Chrome answers on {LOOPBACK}:{port}, but ai-shell's relay did not open " 

631 f"{bind_address}:{port} within {int(CHROME_HOST_PROBE_TIMEOUT_SECONDS)} " 

632 f"seconds. Its log: {relay_log_path(port)}" 

633 ) 

634 

635 if _wait_until_ready( 

636 probe_chrome_port, 

637 container_name, 

638 port, 

639 timeout_seconds=CHROME_CONTAINER_PROBE_TIMEOUT_SECONDS, 

640 ): 

641 logger.info("Chrome ready on port %d for project %s", port, project_name) 

642 return port 

643 

644 raise LocalChromeUnavailable( 

645 f"Chrome is listening on {bind_address}:{port} for project '{project_name}', but the " 

646 "debug port did not become reachable from the dev container within " 

647 f"{int(CHROME_CONTAINER_PROBE_TIMEOUT_SECONDS)} seconds.\n\n" 

648 "Check that the container can reach the host via host.docker.internal " 

649 "(Docker Desktop on Windows/macOS; `host-gateway` on Linux Docker 20.10+ or podman 4.1+)." 

650 )

653def start_chrome_proxy(container_name: str, port: int) -> None: 

654 """Start a TCP proxy inside the container: localhost:<port> -> host.docker.internal:<port>. 

655 

656 Chrome rejects DevTools Protocol requests with a non-localhost Host 

657 header. This proxy lets the MCP server connect to ``localhost:<port>`` 

658 so Chrome sees ``Host: localhost``. 

659 

660 The proxy runs as a detached background process via ``docker exec -d``. 

661 It's idempotent -- if the port is already in use (previous proxy still 

662 running), the new one fails silently and the existing one keeps working. 

663 """ 

664 script = _NODE_PROXY_TEMPLATE.format(port=port) 

665 args = [ 

666 "docker", 

667 "exec", 

668 "-d", 

669 container_name, 

670 "node", 

671 "-e", 

672 script, 

673 ] 

674 logger.debug("Starting Chrome proxy: %s", " ".join(args)) 

675 result = subprocess.run(args, capture_output=True, text=True) 

676 if result.returncode != 0: 

677 logger.warning( 

678 "Chrome proxy start returned %d: %s", result.returncode, result.stderr.strip() 

679 )

682CHROME_MCP_SERVER_NAME = "chrome-devtools" 

683CHROME_MCP_BIN = "chrome-devtools-mcp" 

684CHROME_MCP_PACKAGE = "chrome-devtools-mcp@latest" 

685CHROME_MCP_INSTALL_TIMEOUT_SECONDS = 900 

686CHROME_MCP_PROBE_TIMEOUT_SECONDS = 45.0 

687_MCP_PROTOCOL_VERSION = "2025-06-18"

690def chrome_mcp_server_spec(port: int) -> dict[str, object]: 

691 """MCP server definition for chrome-devtools-mcp against the in-container proxy. 

692 

693 Points at ``localhost:<port>`` (the proxy), not ``host.docker.internal``, 

694 because Chrome rejects the latter's Host header. The command is the 

695 binary installed in the image (``npm install -g``), not ``npx ...@latest``: 

696 an npx fetch on a cold container took longer than Claude Code's MCP 

697 startup timeout, and the server then showed as "failed to start" with 

698 no reason. :func:`ensure_chrome_mcp` installs it on older images. 

699 """ 

700 return { 

701 "command": CHROME_MCP_BIN, 

702 "args": ["--browserUrl", f"http://localhost:{port}"], 

703 }

710def ensure_chrome_mcp(container_name: str) -> None: 

711 """Install ``chrome-devtools-mcp`` globally in the container when the image lacks it.""" 

712 probe = subprocess.run( 

713 ["docker", "exec", container_name, *_login_shell(f"command -v {CHROME_MCP_BIN}")], 

714 capture_output=True, 

715 text=True, 

716 timeout=60, 

717 ) 

718 if probe.returncode == 0: 

719 return 

720 console.print( 

721 f"[dim]Installing the Chrome DevTools MCP server (npm install -g {CHROME_MCP_PACKAGE})...[/dim]" 

722 ) 

723 result = subprocess.run( 

724 ["docker", "exec", container_name, *_login_shell(f"npm install -g {CHROME_MCP_PACKAGE}")], 

725 capture_output=True, 

726 text=True, 

727 timeout=CHROME_MCP_INSTALL_TIMEOUT_SECONDS, 

728 ) 

729 if result.returncode != 0: 

730 raise LocalChromeUnavailable( 

731 "Could not install chrome-devtools-mcp in the dev container, so Claude would " 

732 "have no Chrome tools.\n" 

733 f" npm said: {(result.stderr or result.stdout).strip()[-500:]}" 

734 )

737def _mcp_probe_messages() -> str: 

738 """The three JSON-RPC lines of the probe: handshake, then one real tool call.""" 

739 messages = [ 

740 { 

741 "jsonrpc": "2.0", 

742 "id": 1, 

743 "method": "initialize", 

744 "params": { 

745 "protocolVersion": _MCP_PROTOCOL_VERSION, 

746 "capabilities": {}, 

747 "clientInfo": {"name": "ai-shell", "version": "0"}, 

748 }, 

749 }, 

750 {"jsonrpc": "2.0", "method": "notifications/initialized"}, 

751 { 

752 "jsonrpc": "2.0", 

753 "id": 2, 

754 "method": "tools/call", 

755 "params": {"name": "list_pages", "arguments": {}}, 

756 }, 

757 ] 

758 return "".join(json.dumps(m) + "\n" for m in messages)

761def _mcp_tool_text(result: dict[str, object]) -> str: 

762 content = result.get("content") 

763 if not isinstance(content, list): 

764 return "" 

765 return "\n".join( 

766 str(item.get("text", "")) for item in content if isinstance(item, dict)

770def probe_chrome_mcp( 

771 container_name: str, 

772 port: int, 

773 *, 

774 timeout_seconds: float = CHROME_MCP_PROBE_TIMEOUT_SECONDS, 

775) -> str: 

776 """Start chrome-devtools-mcp once in the container and make it list Chrome's pages. 

777 

778 The server connects to the browser lazily, so a handshake alone proves 

779 nothing; ``list_pages`` exercises the whole path (binary, Node, the 

780 in-container proxy, the DevTools port). Returns the tool's text on 

781 success and raises :class:`LocalChromeUnavailable` with the server's own 

782 output otherwise, so the failure is reported at launch instead of 

783 surfacing inside Claude as a server that "failed to start". 

784 """ 

785 import tempfile 

786 import threading 

787 

788 spec = chrome_mcp_server_spec(port) 

789 command = " ".join(str(part) for part in [spec["command"], *spec["args"]]) # type: ignore[misc] 

790 lines: list[str] = [] 

791 done = threading.Event() 

792 with tempfile.TemporaryFile(mode="w+", encoding="utf-8") as stderr: 

793 proc = subprocess.Popen( 

794 ["docker", "exec", "-i", container_name, *_login_shell(f"exec {command}")], 

795 stdin=subprocess.PIPE, 

796 stdout=subprocess.PIPE, 

797 stderr=stderr, 

798 text=True, 

799 encoding="utf-8", 

800 ) 

801 

802 def _read() -> None: 

803 assert proc.stdout is not None 

804 for line in proc.stdout: 

805 lines.append(line) 

806 try: 

807 message = json.loads(line) 

808 except ValueError: 

809 continue 

810 if isinstance(message, dict) and message.get("id") == 2: 

811 done.set() 

812 return 

813 done.set() 

814 

815 reader = threading.Thread(target=_read, daemon=True) 

816 reader.start() 

817 try: 

818 assert proc.stdin is not None 

819 proc.stdin.write(_mcp_probe_messages()) 

820 proc.stdin.flush() 

821 except (OSError, ValueError): 

822 pass 

823 answered = done.wait(timeout_seconds) 

824 try: 

825 if proc.stdin is not None: 

826 proc.stdin.close() 

827 except OSError: 

828 pass 

829 proc.terminate() 

830 try: 

831 proc.wait(timeout=5) 

832 except subprocess.TimeoutExpired: 

833 proc.kill() 

834 reader.join(timeout=1) 

835 stderr.seek(0) 

836 server_said = stderr.read().strip()[-800:] 

837 

838 reply: dict[str, object] | None = None 

839 for line in lines: 

840 try: 

841 message = json.loads(line) 

842 except ValueError: 

843 continue 

844 if isinstance(message, dict) and message.get("id") == 2: 

845 reply = message 

846 break 

847 

848 detail = f"\n server output: {server_said}" if server_said else "" 

849 if not answered or reply is None: 

850 raise LocalChromeUnavailable( 

851 f"chrome-devtools-mcp did not answer within {int(timeout_seconds)} seconds " 

852 f"(command: {command}).{detail}" 

853 ) 

854 error = reply.get("error") 

855 if isinstance(error, dict): 

856 raise LocalChromeUnavailable( 

857 f"chrome-devtools-mcp refused list_pages: {error.get('message', error)}{detail}" 

858 ) 

859 result = reply.get("result") 

860 if not isinstance(result, dict): 

861 raise LocalChromeUnavailable(f"chrome-devtools-mcp sent no result for list_pages.{detail}") 

862 text = _mcp_tool_text(result) 

863 if result.get("isError"): 

864 raise LocalChromeUnavailable( 

865 f"chrome-devtools-mcp could not reach Chrome on localhost:{port} from the " 

866 f"container: {text or 'no detail'}{detail}" 

867 ) 

868 return text

878 """Write the chrome-devtools-mcp server config JSON. 

879 

880 Returns the path to the written file. The file lives under 

881 ``~/.config/ai-shell/`` by default so it persists across sessions 

882 without polluting the project directory.

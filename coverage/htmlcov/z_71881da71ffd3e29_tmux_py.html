50def sanitize_session_name(name: str) -> str: 

51 """Return a tmux-safe session name (no dots or colons).""" 

52 cleaned = re.sub(r"[^a-zA-Z0-9_-]+", "-", name).strip("-") 

53 return cleaned or "ai-shell"

56def build_tmux_workspace_cmd( 

57 agent_cmd: list[str], 

58 *, 

59 session_name: str, 

60 agent_label: str = "agent", 

61 editor_cmd: str = "nvim", 

62) -> list[str]: 

63 """Build a container command that opens a split tmux workspace. 

64 

65 Layout: 

66 - Left pane: Neovim (drops to a login bash shell if ``:q`` is run) 

67 - Right pane: *agent_cmd* (focused on launch) 

68 

69 If a tmux session named *session_name* already exists in the container, 

70 the command re-attaches to it instead of creating a duplicate session. 

71 """ 

72 session = sanitize_session_name(session_name) 

73 s_q = shlex.quote(session) 

74 label_q = shlex.quote(agent_label) 

75 left_shell = shlex.quote(f"{editor_cmd}; exec bash -l") 

76 agent_str = shlex.join(agent_cmd) 

77 right_shell = shlex.quote(f"exec {agent_str}") 

78 status_hint = shlex.quote(" ^B h/l pane | ^B z zoom | ^B d detach | tmux-help ") 

79 

80 script = ( 

81 f"if ! command -v tmux >/dev/null 2>&1; then " 

82 f"echo '[warn] tmux is not installed in this container; running directly.' >&2; " 

83 f"exec {agent_str}; " 

84 f"fi; " 

85 f"if tmux has-session -t {s_q} 2>/dev/null; then " 

86 f"exec tmux attach-session -t {s_q}; " 

87 f"fi; " 

88 f'LEFT_PANE=$(tmux new-session -d -P -F "#{{pane_id}}" -s {s_q} -c "$PWD" ' 

89 f"bash -lc {left_shell}); " 

90 f'tmux select-pane -t "$LEFT_PANE" -T {shlex.quote(editor_cmd)}; ' 

91 f'RIGHT_PANE=$(tmux split-window -h -P -F "#{{pane_id}}" -t "$LEFT_PANE" -c "$PWD" ' 

92 f"bash -lc {right_shell}); " 

93 f'tmux select-pane -t "$RIGHT_PANE" -T {label_q}; ' 

94 f"tmux set-option -t {s_q} pane-border-status top; " 

95 f"tmux set-option -t {s_q} mouse on; " 

96 f"tmux set-option -t {s_q} status-right {status_hint}; " 

97 f'tmux select-pane -t "$RIGHT_PANE"; ' 

98 f"exec tmux attach-session -t {s_q}" 

99 ) 

100 return ["bash", "-lc", script]
